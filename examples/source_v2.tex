\begin{theorem}Let $K/k$ be a finite field extension. Then the functor $F$ preserves all finite products.\end{theorem}
\begin{theorem}Let $X$ be a space. For every $x$ with $0\leq x\leq1$, the map $F$ is continuous at $x$.\end{theorem}
\begin{theorem}Let $r>0$ be a real parameter. Then $F_r(x)$ converges uniformly for every $x\in X$.\end{theorem}
